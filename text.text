<script>
  function renderReport() {
    const paymentData = JSON.parse(localStorage.getItem("paymentData")) || [];
    const tbody = document.getElementById("payment-information");
    tbody.innerHTML = "";

    paymentData.forEach(info => {
      const row = document.createElement("tr");
      row.innerHTML = `
        <td>${info.card}</td>
        <td>${info.name}</td>
        <td>${info.paid}</td>
        <td>${info.debitAccountNo}</td>
        <td>${info.bdt}</td>
        <td>${info.contact}</td>
        <td>${info.creditAccount}</td>
        <td>${info.branch}</td>
      `;
      tbody.appendChild(row);
    });
  }

  // ✅ Initial render
  window.onload = renderReport;

  // ✅ Listen for updates from index.html
  const paymentChannel = new BroadcastChannel('payment_sync');
  paymentChannel.onmessage = (event) => {
    if (event.data.type === "update") {
      renderReport(); // re-render with latest data
    }
  };
</script>
